
import os
import sys
from pathlib import Path

# Configure Python for Spark
os.environ["PYSPARK_PYTHON"] = sys.executable
os.environ["PYSPARK_DRIVER_PYTHON"] = sys.executable

from pyspark.sql import SparkSession
from read_bronze import read_new_bronze
from deduplicate_news import deduplicate_news
from validate_news import validate_news
from clean_news import clean_news


spark = (
    SparkSession.builder
    .master("local[*]")
    .appName("TestActualNewsCleaning")
    .getOrCreate()
)

spark.sparkContext.setLogLevel("ERROR")

try:
    # 1. Read actual Bronze records
    df, files = read_new_bronze(spark)

    print("\nBronze records:", df.count())

    # 2. Deduplicate
    unique, missing = deduplicate_news(df)

    candidates = unique.unionByName(
        missing,
        allowMissingColumns=True
    )

    # 3. Validate
    valid, invalid = validate_news(candidates)

    print("Valid records:", valid.count())
    print("Invalid records:", invalid.count())

    # 4. Clean valid records
    cleaned = clean_news(valid)

    print("\n========== CLEANED NEWS ==========")

    cleaned.select(
        "headline",
        "cleaned_full_text",
        "text_for_analysis"
    ).show(20, truncate=False)

    print("\n========== CLEANING CHECKS ==========")

    print("Cleaned records:", cleaned.count())

    print(
        "Empty analysis text:",
        cleaned.filter(
            "text_for_analysis IS NULL OR trim(text_for_analysis) = ''"
        ).count()
    )

finally:
    spark.stop()
    print("\nSpark session stopped.")
