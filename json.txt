
import os
import sys

# --------------------------------------------------
# 1. CONFIGURE PYTHON FOR SPARK
# --------------------------------------------------

os.environ["PYSPARK_PYTHON"] = sys.executable
os.environ["PYSPARK_DRIVER_PYTHON"] = sys.executable

from pyspark.sql import SparkSession

from read_bronze import read_new_bronze
from deduplicate_news import deduplicate_news
from validate_news import validate_news
from clean_news import clean_news


# --------------------------------------------------
# 2. CREATE SPARK SESSION
# --------------------------------------------------

spark = (
    SparkSession.builder
    .appName("IncrementalSilverNewsPipeline")
    .master("local[*]")
    .config("spark.sql.session.timeZone", "UTC")
    .getOrCreate()
)

spark.sparkContext.setLogLevel("ERROR")


try:

    print("\n========================================")
    print("       SILVER NEWS PIPELINE")
    print("========================================")

    # --------------------------------------------------
    # 3. FILE-LEVEL INCREMENTAL BRONZE READING
    # --------------------------------------------------

    news_df, new_files = read_new_bronze(spark)

    if news_df is None:

        print("\nNo records available for processing.")

    else:

        # --------------------------------------------------
        # 4. ARTICLE-LEVEL DEDUPLICATION
        # --------------------------------------------------

        unique_news_df, missing_id_df = deduplicate_news(news_df)

        # Include missing-ID records for validation and DLQ.
        candidates_df = unique_news_df.unionByName(
            missing_id_df,
            allowMissingColumns=True
        )

        print("\n========== DEDUPLICATED CANDIDATE RECORDS ==========")

        candidates_df.select(
            "article_id",
            "headline",
            "bronze_file"
        ).show(truncate=False)

        # --------------------------------------------------
        # 5. VALIDATION
        # --------------------------------------------------

        valid_df, invalid_df = validate_news(candidates_df)

        print("\n========== VALIDATION RESULTS ==========")

        print("Valid records:", valid_df.count())
        print("Invalid records:", invalid_df.count())

        # --------------------------------------------------
        # 6. HTML CLEANING
        # --------------------------------------------------

        cleaned_df = clean_news(valid_df)

        print("\n========== CLEANED NEWS ==========")

        cleaned_df.select(
            "article_id",
            "headline",
            "cleaned_headline",
            "full_text",
            "cleaned_full_text",
            "text_for_analysis"
        ).show(truncate=False)

        # --------------------------------------------------
        # 7. DEAD-LETTER DISPLAY
        # --------------------------------------------------

        print("\n========== DEAD-LETTER RECORDS ==========")

        invalid_df.select(
            "article_id",
            "validation_status",
            "validation_reason",
            "bronze_file",
            "topic",
            "partition",
            "offset",
            "raw_json"
        ).show(truncate=False)

        # --------------------------------------------------
        # 8. PIPELINE SUMMARY
        # --------------------------------------------------

        print("\n========== PIPELINE SUMMARY ==========")

        print("New Bronze files discovered:", len(new_files))
        print("New unique articles:", unique_news_df.count())
        print("Records with missing article IDs:", missing_id_df.count())
        print("Valid records:", valid_df.count())
        print("Invalid records:", invalid_df.count())
        print("Cleaned records:", cleaned_df.count())

        print("\nBRONZE READING, DEDUPLICATION, VALIDATION")
        print("AND HTML CLEANING COMPLETED!")

        # IMPORTANT:
        # Parquet output persistence and checkpoint commits
        # will be implemented after all Silver stages
        # are integrated and tested.

finally:

    spark.stop()

    print("\nSPARK SESSION STOPPED")
