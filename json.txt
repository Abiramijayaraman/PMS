import sys
from pathlib import Path
 
from pyspark.sql import SparkSession
from pyspark.sql.functions import (
    col,
    from_json,
    trim,
    coalesce,
    lit,
    when,
    concat_ws,
)
from pyspark.sql.types import StructType, StructField, StringType
 
import pyarrow.parquet as pq
 
 
# --------------------------------------------------
# 1. Project paths
# --------------------------------------------------
 
PROJECT_ROOT = Path(__file__).resolve().parents[2]
 
BRONZE_PATH = (
    PROJECT_ROOT
    / "silver_news"
    / "data"
    / "simulated_bronze"
    / "news"
)
 
 
# --------------------------------------------------
# 2. Configure Python for Spark
# --------------------------------------------------
 
python_executable = sys.executable
 
import os
os.environ["PYSPARK_PYTHON"] = python_executable
os.environ["PYSPARK_DRIVER_PYTHON"] = python_executable
 
 
# --------------------------------------------------
# 3. Create Spark session
# --------------------------------------------------
 
spark = (
    SparkSession.builder
    .appName("SilverNewsValidation")
    .master("local[*]")
    .getOrCreate()
)
 
spark.sparkContext.setLogLevel("ERROR")
 
 
# --------------------------------------------------
# 4. Read Bronze Parquet using PyArrow
# --------------------------------------------------
 
print("\n========== READING BRONZE DATA ==========")
 
if not BRONZE_PATH.exists():
    raise FileNotFoundError(
        f"Bronze Parquet folder not found: {BRONZE_PATH}"
    )
 
table = pq.read_table(str(BRONZE_PATH))
bronze_records = table.to_pylist()
 
print(f"Total Bronze records: {len(bronze_records)}")
 
 
# We only need raw_json for this validation step.
# This also avoids timestamp conversion issues on Windows.
 
raw_records = [
    {"raw_json": record.get("raw_json")}
    for record in bronze_records
]
 
bronze_df = spark.createDataFrame(raw_records)
 
print("\nBronze input records:")
bronze_df.show(truncate=False)
 
 
# --------------------------------------------------
# 5. Define the producer's JSON schema
# --------------------------------------------------
 
article_schema = StructType([
    StructField("article_id", StringType(), True),
    StructField("published_at", StringType(), True),
    StructField("source", StringType(), True),
    StructField("headline", StringType(), True),
    StructField("full_text", StringType(), True),
    StructField("body_html", StringType(), True),  # Legacy sample support
    StructField("url", StringType(), True),
    StructField("ingested_at", StringType(), True),
])
 
 
# --------------------------------------------------
# 6. Parse raw JSON
# --------------------------------------------------
 
print("\n========== PARSING NEWS JSON ==========")
 
parsed_df = bronze_df.withColumn(
    "article",
    from_json(col("raw_json"), article_schema)
)
 
news_df = parsed_df.select(
    col("article.article_id").alias("article_id"),
    col("article.published_at").alias("published_at"),
    col("article.source").alias("source"),
    col("article.headline").alias("headline"),
    coalesce(
        col("article.full_text"),
        col("article.body_html")
    ).alias("full_text"),
    col("article.url").alias("url"),
    col("article.ingested_at").alias("ingested_at"),
    col("article").isNull().alias("json_parse_failed"),
)
 
 
# --------------------------------------------------
# 7. Validate required fields
# --------------------------------------------------
 
print("\n========== VALIDATING NEWS ==========")
 
# Required fields:
# 1. article_id
# 2. url
#
# At least one usable text field is required for
# downstream sentiment analysis:
# headline OR full_text.
#
# published_at and source are optional.
 
has_article_id = (
    col("article_id").isNotNull()
    & (trim(col("article_id")) != "")
)
 
has_url = (
    col("url").isNotNull()
    & (trim(col("url")) != "")
)
 
has_headline = (
    col("headline").isNotNull()
    & (trim(col("headline")) != "")
)
 
has_full_text = (
    col("full_text").isNotNull()
    & (trim(col("full_text")) != "")
)
 
has_content = has_headline | has_full_text
 
 
# Build a reason for every invalid record.
validation_reason = concat_ws(
    "; ",
    when(col("json_parse_failed"), lit("invalid_json")),
    when(~has_article_id, lit("missing_article_id")),
    when(~has_url, lit("missing_url")),
    when(~has_content, lit("missing_headline_and_full_text")),
)
 
validated_df = news_df.withColumn(
    "validation_reason",
    validation_reason
).withColumn(
    "validation_status",
    when(col("validation_reason") == "", "VALID")
    .otherwise("INVALID")
)
 
 
# --------------------------------------------------
# 8. Separate valid and invalid records
# --------------------------------------------------
 
valid_news_df = validated_df.filter(
    col("validation_status") == "VALID"
)
 
invalid_news_df = validated_df.filter(
    col("validation_status") == "INVALID"
)
 
 
# --------------------------------------------------
# 9. Display validation results
# --------------------------------------------------
 
total_count = validated_df.count()
valid_count = valid_news_df.count()
invalid_count = invalid_news_df.count()
 
print("\n========== VALIDATION SUMMARY ==========")
 
print(f"Total records  : {total_count}")
print(f"Valid records  : {valid_count}")
print(f"Invalid records: {invalid_count}")
 
 
print("\n========== VALID NEWS ==========")
 
valid_news_df.select(
    "article_id",
    "headline",
    "published_at",
    "validation_status"
).show(truncate=False)
 
 
print("\n========== INVALID NEWS / DEAD-LETTER CANDIDATES ==========")
 
invalid_news_df.select(
    "article_id",
    "url",
    "validation_reason",
    "validation_status"
).show(truncate=False)
 
 
# --------------------------------------------------
# 10. Stop Spark
# --------------------------------------------------
 
spark.stop()
 
print("\nNews validation completed successfully!")
