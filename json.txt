
import os
import sys

# Configure Python for Spark.
os.environ["PYSPARK_PYTHON"] = sys.executable
os.environ["PYSPARK_DRIVER_PYTHON"] = sys.executable

from pyspark.sql import SparkSession

from read_bronze import read_new_bronze
from deduplicate_news import deduplicate_news
from validate_news import validate_news


# --------------------------------------------------
# 1. CREATE SPARK SESSION
# --------------------------------------------------

spark = (
    SparkSession.builder
    .appName("IncrementalSilverNewsPipeline")
    .master("local[*]")
    .config("spark.sql.session.timeZone", "UTC")
    .getOrCreate()
)

spark.sparkContext.setLogLevel("ERROR")


try:

    print("\n========================================")
    print("       SILVER NEWS PIPELINE")
    print("========================================")

    # --------------------------------------------------
    # 2. FILE-LEVEL INCREMENTAL BRONZE READING
    # --------------------------------------------------

    news_df, new_files = read_new_bronze(spark)

    if news_df is None:

        print("\nNo records available for processing.")

    else:

        # --------------------------------------------------
        # 3. ARTICLE-LEVEL DEDUPLICATION
        # --------------------------------------------------

        unique_news_df, missing_id_df = deduplicate_news(news_df)

        # Keep missing-ID records for validation.
        # They must not be discarded before DLQ handling.
        candidates_df = unique_news_df.unionByName(
            missing_id_df,
            allowMissingColumns=True
        )

        print("\n========== CANDIDATE RECORDS ==========")

        candidates_df.show(truncate=False)

        # --------------------------------------------------
        # 4. VALIDATION
        # --------------------------------------------------

        valid_df, invalid_df = validate_news(candidates_df)

        # --------------------------------------------------
        # 5. DISPLAY VALID RECORDS
        # --------------------------------------------------

        print("\n========== VALID NEWS RECORDS ==========")

        valid_df.show(truncate=False)

        # --------------------------------------------------
        # 6. DISPLAY DEAD-LETTER RECORDS
        # --------------------------------------------------

        print("\n========== DEAD-LETTER RECORDS ==========")

        invalid_df.select(
            "article_id",
            "url",
            "validation_status",
            "validation_reason",
            "bronze_file",
            "topic",
            "partition",
            "offset",
            "raw_json"
        ).show(truncate=False)

        # --------------------------------------------------
        # 7. PIPELINE SUMMARY
        # --------------------------------------------------

        print("\n========== PIPELINE SUMMARY ==========")

        print("New Bronze files discovered:", len(new_files))
        print("New unique articles:", unique_news_df.count())
        print("Records with missing article IDs:", missing_id_df.count())
        print("Valid records:", valid_df.count())
        print("Invalid records:", invalid_df.count())

        print("\nBRONZE READING, DEDUPLICATION AND VALIDATION COMPLETED!")

        # IMPORTANT:
        # Checkpoint updates and final Parquet writes will be added
        # after the complete Silver processing stages are integrated.
        # Do not mark files or article IDs as processed yet.

finally:

    spark.stop()

    print("\nSPARK SESSION STOPPED")
