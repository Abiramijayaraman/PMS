import sys
from pathlib import Path
 
import torch
from transformers import (
    AutoTokenizer,
    AutoModelForSequenceClassification
)
 
 
# --------------------------------------------------
# 1. FIND THE LOCAL FINBERT MODEL
# --------------------------------------------------
 
PROJECT_FOLDER = Path(__file__).resolve().parents[1]
 
MODEL_PATH = PROJECT_FOLDER / "models" / "finbert"
 
print("FinBERT local model path:", MODEL_PATH)
 
 
# --------------------------------------------------
# 2. CHECK WHETHER ALL REQUIRED FILES EXIST
# --------------------------------------------------
 
required_files = [
    "config.json",
    "model.safetensors",
    "tokenizer_config.json",
    "special_tokens_map.json",
    "vocab.txt"
]
 
missing_files = [
    filename
    for filename in required_files
    if not (MODEL_PATH / filename).exists()
]
 
if missing_files:
    print("\nERROR: Required model files are missing:")
 
    for filename in missing_files:
        print("-", filename)
 
    sys.exit(1)
 
 
print("\nAll required FinBERT files are available.")
 
 
# --------------------------------------------------
# 3. SELECT THE COMPUTING DEVICE
# --------------------------------------------------
 
DEVICE = torch.device(
    "cuda" if torch.cuda.is_available() else "cpu"
)
 
print("\nUsing device:", DEVICE)
 
 
# --------------------------------------------------
# 4. LOAD FINBERT LOCALLY
# --------------------------------------------------
 
print("\nLoading FinBERT tokenizer...")
 
tokenizer = AutoTokenizer.from_pretrained(
    str(MODEL_PATH),
    local_files_only=True
)
 
print("Tokenizer loaded successfully.")
 
print("\nLoading FinBERT model...")
 
model = AutoModelForSequenceClassification.from_pretrained(
    str(MODEL_PATH),
    local_files_only=True
)
 
model.to(DEVICE)
model.eval()
 
print("FinBERT model loaded successfully!")
 
 
# --------------------------------------------------
# 5. READ SENTIMENT LABELS FROM MODEL CONFIG
# --------------------------------------------------
 
id2label = model.config.id2label
 
print("\nModel sentiment labels:")
print(id2label)
 
 
# --------------------------------------------------
# 6. SENTIMENT ANALYSIS FUNCTION
# --------------------------------------------------
 
def analyse_sentiment(news_text):
 
    # Convert news text into tokens
    inputs = tokenizer(
        news_text,
        return_tensors="pt",
        truncation=True,
        max_length=512
    )
 
    # Move input tensors to the selected device
    inputs = {
        key: value.to(DEVICE)
        for key, value in inputs.items()
    }
 
    # Disable gradient calculation during prediction
    with torch.no_grad():
 
        outputs = model(**inputs)
 
        # Convert model logits into probabilities
        probabilities = torch.softmax(
            outputs.logits,
            dim=-1
        )[0]
 
    # Associate each probability with its actual label
    scores = {
        str(id2label[index]).lower(): probabilities[index].item()
        for index in range(len(probabilities))
    }
 
    # Identify the highest-probability sentiment
    sentiment = max(scores, key=scores.get)
 
    positive_probability = scores.get("positive", 0.0)
    negative_probability = scores.get("negative", 0.0)
    neutral_probability = scores.get("neutral", 0.0)
 
    # Calculate sentiment score
    sentiment_score = (
        positive_probability - negative_probability
    )
 
    return {
        "sentiment": sentiment,
        "positive_probability": positive_probability,
        "negative_probability": negative_probability,
        "neutral_probability": neutral_probability,
        "sentiment_score": sentiment_score
    }
 
 
# --------------------------------------------------
# 7. TEST WITH SAMPLE FINANCIAL NEWS
# --------------------------------------------------
 
sample_news = [
 
    "Reliance Industries reports strong quarterly profits.",
 
    "TCS faces declining revenue and weak market demand.",
 
    "HDFC Bank announces its quarterly financial results.",
 
    "Infosys shares rise after announcing a major AI investment.",
 
    "stocks to watch today: SBI",
]
 
 
# --------------------------------------------------
# 8. DISPLAY PREDICTIONS
# --------------------------------------------------
 
print("\n" + "=" * 60)
print("FINBERT FINANCIAL NEWS SENTIMENT ANALYSIS")
print("=" * 60)
 
for index, news in enumerate(sample_news, start=1):
 
    result = analyse_sentiment(news)
 
    print(f"\nARTICLE {index}")
    print("-" * 60)
 
    print("News:", news)
 
    print("Predicted sentiment:", result["sentiment"])
 
    print(
        "Positive probability:",
        round(result["positive_probability"], 4)
    )
 
    print(
        "Negative probability:",
        round(result["negative_probability"], 4)
    )
 
    print(
        "Neutral probability:",
        round(result["neutral_probability"], 4)
    )
 
    print(
        "Sentiment score:",
        round(result["sentiment_score"], 4)
    )
 
print("\n" + "=" * 60)
print("FINBERT TEST COMPLETED")
print("=" * 60)
