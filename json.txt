
from pyspark.sql.functions import (
    col,
    regexp_replace,
    trim,
    coalesce,
    lit,
    when
)


# --------------------------------------------------
# 1. HTML CLEANING
# --------------------------------------------------

def clean_news(news_df):

    print("\n========== STARTING HTML CLEANING ==========")

    # Clean HTML from full_text.
    cleaned_df = news_df.withColumn(
        "cleaned_full_text",
        regexp_replace(
            coalesce(col("full_text"), lit("")),
            r"<[^>]*>",
            " "
        )
    )

    # Replace common HTML entities.
    cleaned_df = cleaned_df.withColumn(
        "cleaned_full_text",
        regexp_replace(
            col("cleaned_full_text"),
            r"(?i)&nbsp;|&#160;|&amp;|&quot;|&#39;|&lt;|&gt;",
            " "
        )
    )

    # Normalize whitespace.
    cleaned_df = cleaned_df.withColumn(
        "cleaned_full_text",
        trim(
            regexp_replace(
                col("cleaned_full_text"),
                r"\s+",
                " "
            )
        )
    )

    # Clean headline separately.
    cleaned_df = cleaned_df.withColumn(
        "cleaned_headline",
        trim(
            regexp_replace(
                coalesce(col("headline"), lit("")),
                r"\s+",
                " "
            )
        )
    )

    # Prepare text for FinBERT.
    # Prefer headline; use cleaned full_text as fallback.
    cleaned_df = cleaned_df.withColumn(
        "text_for_analysis",
        when(
            col("cleaned_headline") != "",
            col("cleaned_headline")
        ).otherwise(col("cleaned_full_text"))
    )

    print("Input records:", news_df.count())
    print("HTML cleaning completed.")

    return cleaned_df
