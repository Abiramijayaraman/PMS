
import json
from pathlib import Path

from pyspark.sql.functions import (
    col,
    trim,
    row_number,
    to_timestamp,
    desc_nulls_last,
    asc
)
from pyspark.sql.window import Window


# --------------------------------------------------
# 1. PROJECT PATHS
# --------------------------------------------------

PROJECT_ROOT = Path(__file__).resolve().parents[2]

CHECKPOINT_DIR = (
    PROJECT_ROOT
    / "silver_news"
    / "data"
    / "checkpoints"
)

PROCESSED_IDS_PATH = (
    CHECKPOINT_DIR
    / "processed_article_ids.json"
)


# --------------------------------------------------
# 2. LOAD PREVIOUSLY PROCESSED ARTICLE IDS
# --------------------------------------------------

def load_processed_article_ids():

    if not PROCESSED_IDS_PATH.exists():
        return set()

    with open(
        PROCESSED_IDS_PATH,
        "r",
        encoding="utf-8"
    ) as file:

        data = json.load(file)

    return set(data.get("processed_article_ids", []))


# --------------------------------------------------
# 3. ARTICLE-LEVEL DEDUPLICATION
# --------------------------------------------------

def deduplicate_news(news_df):

    print("\n========== ARTICLE-LEVEL DEDUPLICATION ==========")

    # Load article IDs processed in previous successful runs
    processed_ids = load_processed_article_ids()

    print("Previously processed article IDs:", len(processed_ids))

    # Remove records with missing or empty article IDs
    # These records will be handled by validation / DLQ.
    identifiable_df = news_df.filter(
        col("article_id").isNotNull()
        & (trim(col("article_id")) != "")
    )

    missing_id_df = news_df.filter(
        col("article_id").isNull()
        | (trim(col("article_id")) == "")
    )

    # Remove articles already processed in earlier batches
    if processed_ids:

        new_df = identifiable_df.filter(
            ~col("article_id").isin(list(processed_ids))
        )

        previously_processed_df = identifiable_df.filter(
            col("article_id").isin(list(processed_ids))
        )

    else:

        new_df = identifiable_df
        previously_processed_df = identifiable_df.limit(0)

    # Remove duplicate article IDs within the current batch
    window_spec = (
        Window
        .partitionBy("article_id")
        .orderBy(
            desc_nulls_last(
                to_timestamp(col("ingested_at"))
            ),
            asc(col("url")),
            asc(col("headline"))
        )
    )

    unique_new_df = (
        new_df
        .withColumn(
            "row_number",
            row_number().over(window_spec)
        )
        .filter(col("row_number") == 1)
        .drop("row_number")
    )

    # Display results
    total_count = news_df.count()
    missing_id_count = missing_id_df.count()
    previously_processed_count = previously_processed_df.count()
    current_batch_count = new_df.count()
    unique_count = unique_new_df.count()

    print("\nDeduplication Results")
    print("---------------------")
    print("Incoming records:", total_count)
    print("Missing article IDs:", missing_id_count)
    print("Already processed articles:", previously_processed_count)
    print("Duplicate IDs within current batch:", current_batch_count - unique_count)
    print("New unique articles:", unique_count)

    return unique_new_df, missing_id_df


# IMPORTANT:
# This module does not save article IDs to the checkpoint.
# The final pipeline runner will update the checkpoint
# only after all outputs and DLQ records are written successfully.
