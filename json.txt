python -c "from pyspark.sql import SparkSession; from pyspark.sql.functions import *; from pyspark.sql.types import *; s=SparkSession.builder.master('local[*]').appName('DeadLetterTest').getOrCreate(); s.sparkContext.setLogLevel('ERROR'); schema=StructType([StructField('article_id',StringType()),StructField('headline',StringType()),StructField('full_text',StringType()),StructField('url',StringType())]); data=[('TEST-001','Reliance reports profits','Strong results','https://example.com/1'),('', 'SBI shares decline','Shares declined','https://example.com/2'),('TEST-003','Infosys expands','Expansion news',''),('TEST-004','','','https://example.com/4')]; d=s.createDataFrame(data,schema); a=col('article_id').isNotNull() & (trim(col('article_id'))!=''); u=col('url').isNotNull() & (trim(col('url'))!=''); h=col('headline').isNotNull() & (trim(col('headline'))!=''); f=col('full_text').isNotNull() & (trim(col('full_text'))!=''); r=concat_ws('; ',when(~a,'missing_article_id'),when(~u,'missing_url'),when(~(h|f),'missing_headline_and_full_text')); d.withColumn('validation_reason',r).withColumn('validation_status',when(r=='','VALID').otherwise('INVALID')).show(truncate=False); s.stop()"
