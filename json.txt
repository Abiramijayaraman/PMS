
import os
import sys
import json
from pathlib import Path
from datetime import timezone

import pyarrow.parquet as pq

# Configure Python for Spark
os.environ["PYSPARK_PYTHON"] = sys.executable
os.environ["PYSPARK_DRIVER_PYTHON"] = sys.executable

from pyspark.sql import SparkSession
from pyspark.sql.functions import col, from_json
from pyspark.sql.types import (
    StructType,
    StructField,
    StringType,
    IntegerType,
    LongType,
    TimestampType
)

# --------------------------------------------------
# 1. Define paths
# --------------------------------------------------

PROJECT_ROOT = Path(__file__).resolve().parents[2]

BRONZE_PATH = (
    PROJECT_ROOT
    / "silver_news"
    / "data"
    / "actual_bronze"
    / "news"
)

CHECKPOINT_DIR = (
    PROJECT_ROOT
    / "silver_news"
    / "data"
    / "checkpoints"
)

PROCESSED_FILES_PATH = CHECKPOINT_DIR / "processed_files.json"


# --------------------------------------------------
# 2. Checkpoint functions
# --------------------------------------------------

def load_processed_files():

    if not PROCESSED_FILES_PATH.exists():
        return set()

    with open(PROCESSED_FILES_PATH, "r", encoding="utf-8") as file:
        data = json.load(file)

    return set(data.get("processed_files", []))


def discover_new_files():

    if not BRONZE_PATH.exists():
        raise FileNotFoundError(
            f"Bronze folder not found: {BRONZE_PATH}"
        )

    all_files = sorted(BRONZE_PATH.glob("*.parquet"))

    processed_files = load_processed_files()

    new_files = [
        file for file in all_files
        if file.name not in processed_files
    ]

    print("\n========== INCREMENTAL FILE CHECK ==========")

    print("Total Bronze files:", len(all_files))
    print("Previously processed files:", len(processed_files))
    print("New files to process:", len(new_files))

    for file in new_files:
        print("NEW:", file.name)

    for file in all_files:
        if file.name in processed_files:
            print("SKIPPED:", file.name)

    return new_files


def save_processed_files(processed_file_names):

    CHECKPOINT_DIR.mkdir(parents=True, exist_ok=True)

    existing_files = load_processed_files()

    existing_files.update(processed_file_names)

    checkpoint_data = {
        "processed_files": sorted(existing_files)
    }

    temp_path = PROCESSED_FILES_PATH.with_suffix(".tmp")

    with open(temp_path, "w", encoding="utf-8") as file:
        json.dump(checkpoint_data, file, indent=4)

    temp_path.replace(PROCESSED_FILES_PATH)

    print("File checkpoint updated successfully.")


# --------------------------------------------------
# 3. Define schemas
# --------------------------------------------------

bronze_schema = StructType([
    StructField("kafka_key", StringType(), True),
    StructField("raw_json", StringType(), False),
    StructField("topic", StringType(), False),
    StructField("partition", IntegerType(), False),
    StructField("offset", LongType(), False),
    StructField("kafka_ts", TimestampType(), False),
    StructField("bronze_file", StringType(), True)
])

article_schema = StructType([
    StructField("article_id", StringType(), True),
    StructField("published_at", StringType(), True),
    StructField("source", StringType(), True),
    StructField("headline", StringType(), True),
    StructField("full_text", StringType(), True),
    StructField("url", StringType(), True),
    StructField("ingested_at", StringType(), True)
])


# --------------------------------------------------
# 4. Create Spark session
# --------------------------------------------------

spark = (
    SparkSession.builder
    .appName("IncrementalBronzeNewsReader")
    .master("local[*]")
    .config("spark.sql.session.timeZone", "UTC")
    .getOrCreate()
)

spark.sparkContext.setLogLevel("ERROR")

try:

    # --------------------------------------------------
    # 5. Discover only new Bronze files
    # --------------------------------------------------

    print("\n========== INCREMENTAL BRONZE READER ==========")

    new_files = discover_new_files()

    if not new_files:
        print("\nNo new Bronze files. Nothing to process.")

    else:

        # --------------------------------------------------
        # 6. Read new Parquet files
        # --------------------------------------------------

        records = []

        for parquet_file in new_files:

            print(f"\nReading: {parquet_file.name}")

            arrow_table = pq.read_table(parquet_file)
            file_records = arrow_table.to_pylist()

            for record in file_records:

                # Track the source file for every record
                record["bronze_file"] = parquet_file.name

                # Convert timezone-aware timestamps to naive UTC
                timestamp = record["kafka_ts"]

                if timestamp is not None and timestamp.tzinfo is not None:
                    record["kafka_ts"] = (
                        timestamp.astimezone(timezone.utc)
                        .replace(tzinfo=None)
                    )

            print("Records in file:", len(file_records))

            records.extend(file_records)

        print("\nNew Bronze files:", len(new_files))
        print("New Bronze records:", len(records))

        # --------------------------------------------------
        # 7. Create Spark DataFrame
        # --------------------------------------------------

        bronze_df = spark.createDataFrame(
            records,
            schema=bronze_schema
        )

        print("\nBRONZE DATA SCHEMA:")
        bronze_df.printSchema()

        # --------------------------------------------------
        # 8. Parse raw JSON
        # --------------------------------------------------

        print("\n========== PARSING RAW JSON ==========")

        parsed_df = bronze_df.withColumn(
            "article",
            from_json(col("raw_json"), article_schema)
        )

        # --------------------------------------------------
        # 9. Extract article fields
        # --------------------------------------------------

        news_df = parsed_df.select(
            col("bronze_file"),
            col("kafka_key"),
            col("topic"),
            col("partition"),
            col("offset"),
            col("kafka_ts"),
            col("article.article_id").alias("article_id"),
            col("article.published_at").alias("published_at"),
            col("article.source").alias("source"),
            col("article.headline").alias("headline"),
            col("article.full_text").alias("full_text"),
            col("article.url").alias("url"),
            col("article.ingested_at").alias("ingested_at")
        )

        # --------------------------------------------------
        # 10. Display extracted records
        # --------------------------------------------------

        print("\n========== NEW EXTRACTED NEWS ==========")

        news_df.show(truncate=False)

        print("\nEXTRACTED NEWS SCHEMA:")
        news_df.printSchema()

        print("\nTOTAL EXTRACTED ARTICLES:", news_df.count())

        # --------------------------------------------------
        # 11. Basic data checks
        # --------------------------------------------------

        print("\n========== BASIC DATA CHECKS ==========")

        print(
            "Missing headlines:",
            news_df.filter(col("headline").isNull()).count()
        )

        print(
            "Missing full_text:",
            news_df.filter(col("full_text").isNull()).count()
        )

        print(
            "Missing published_at:",
            news_df.filter(col("published_at").isNull()).count()
        )

        print("\nINCREMENTAL BRONZE READING COMPLETED!")

        # IMPORTANT:
        # Do not update the processed-file checkpoint here.
        # The final pipeline runner will update it only after
        # successful validation, DLQ and Silver output writes.

finally:

    spark.stop()
    print("\nSPARK SESSION STOPPED")
